// スプレッドシートのデータをオブジェクトの配列に変換する共通関数 function getSheetData(sheetName) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(sheetName); var rows = sheet.getDataRange().getValues(); var headers = rows[0]; var data = []; for (var i = 1; i < rows.length; i++) { var row = rows[i]; var obj = {}; for (var j = 0; j < headers.length; j++) { // 日付オブジェクトの場合は、文字列(YYYY-MM-DD)に安全に変換 if (row[j] instanceof Date) { var d = row[j]; obj[headers[j]] = d.getFullYear() + '-' + ('0' + (d.getMonth() + 1)).slice(-2) + '-' + ('0' + d.getDate()).slice(-2); } else { obj[headers[j]] = row[j]; } } data.push(obj); } return data; } // データの取得（GETリクエスト） function doGet(e) { var sheetName = e.parameter.sheet; var callback = e.parameter.callback; var data = getSheetData(sheetName); var responseText = callback + "(" + JSON.stringify({ data: data }) + ");"; return ContentService.createTextOutput(responseText).setMimeType(ContentService.MimeType.JAVASCRIPT); } // データの更新・保存・削除（POSTリクエスト） function doPost(e) { var postData = JSON.parse(e.postData.contents); var sheetName = postData.sheet; var action = postData.action; var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName(sheetName); if (action === 'upsert') { var id = postData.id; var rowData = postData.row; var rows = sheet.getDataRange().getValues(); var foundIndex = -1; // 既存のIDがあるかチェック for (var i = 1; i < rows.length; i++) { if (rows[i][0] == id) { foundIndex = i + 1; // スプレッドシートの行番号(2から開始) break; } } // シートごとに列の並び順を設定（新しい項目構成に対応） var newRow = []; if (sheetName === 'routes') { newRow = [rowData.id, rowData.name, rowData.time, rowData.vehicleId, rowData.days, rowData.note]; } else if (sheetName === 'vehicles') { newRow = [rowData.id, rowData.name, rowData.type, rowData.shaken, rowData.insurance, rowData.note]; } if (foundIndex > -1) { // 上書き sheet.getRange(foundIndex, 1, 1, newRow.length).setValues([newRow]); } else { // 新規追加 sheet.appendRow(newRow); } } else if (action === 'delete') { var id = postData.id; var rows = sheet.getDataRange().getValues(); for (var i = 1; i < rows.length; i++) { if (rows[i][0] == id) { sheet.deleteRow(i + 1); break; } } } else if (action === 'save_check') { // 点検表用の保存ロジック（既存の月データがあれば上書き、なければ追加） var vId = postData.vehicleId; var year = postData.year; var month = postData.month; var marks = postData.marks; var note = postData.note; var savedAt = postData.savedAt; var rows = sheet.getDataRange().getValues(); // まず、その車両・年月の既存データを削除（一括上書きするため） for (var i = rows.length - 1; i >= 1; i--) { if (rows[i][0] == vId && rows[i][1] == year && rows[i][2] == month) { sheet.deleteRow(i + 1); } } // 新しい点検結果を1項目1行で追加 for (var key in marks) { if (marks.hasOwnProperty(key)) { sheet.appendRow([vId, year, month, key, marks[key], note, savedAt]); } } } return ContentService.createTextOutput(JSON.stringify({ status: 'success' })).setMimeType(ContentService.MimeType.JSON); }
