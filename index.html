```javascript function doGet(e) {   var sheetName = e.parameter.sheet;   var callback = e.parameter.callback;      var ss = SpreadsheetApp.getActiveSpreadsheet();   var sheet = ss.getSheetByName(sheetName);      var rows = sheet.getDataRange().getValues();   var headers = rows[0];   var data = [];      for (var i = 1; i < rows.length; i++) {     var row = rows[i];     var record = {};     for (var j = 0; j < headers.length; j++) {       if (headers[j] === "note" || headers[j] === "shaken" || headers[j] === "insurance") {         if (row[j] instanceof Date) {           record[headers[j]] = Utilities.formatDate(row[j], Session.getScriptTimeZone(), "yyyy-MM-dd");         } else {           record[headers[j]] = row[j];         }       } else {         record[headers[j]] = row[j];       }     }     data.push(record);   }      var result = JSON.stringify({ data: data });   var sheet = ss.getSheetByName(params.sheet);      if (params.action === 'upsert') {     var data = sheet.getDataRange().getValues();     var headers = data[0];     var idIndex = headers.indexOf('id');     var rowIndex = -1;          for (var i = 1; i < data.length; i++) {       if (data[i][idIndex] == params.id) {         rowIndex = i + 1;         break;       }     }          var rowData = [];     for (var j = 0; j < headers.length; j++) {       rowData.push(params.row[headers[j]] || '');     }          if (rowIndex > 0) {       sheet.getRange(rowIndex, 1, 1, headers.length).setValues([rowData]);     } else {       sheet.appendRow(rowData);     }   } else if (params.action === 'delete') {     var data = sheet.getDataRange().getValues();     var idIndex = data[0].indexOf('id');     for (var i = 1; i < data.length; i++) {       if (data[i][idIndex] == params.id) {         sheet.deleteRow(i + 1);         break;       }     }   } else if (params.action === 'save_check') {     var data = sheet.getDataRange().getValues();     var vIdx = data[0].indexOf('vehicleId');     var yIdx = data[0].indexOf('year');     var mIdx = data[0].indexOf('month');     var kIdx = data[0].indexOf('itemKey');          for (var key in params.marks) {       var existRow = -1;       for (var i = 1; i < data.length; i++) {         if (data[i][vIdx] == params.vehicleId && data[i][yIdx] == params.year && data[i][mIdx] == params.month && data[i][kIdx] == key) {           existRow = i + 1;           break;         }       }              var rowData = [];       var headers = data[0];       var record = {         vehicleId: params.vehicleId,         year: params.year,         month: params.month,         itemKey: key,         value: params.marks[key],         note: params.note,         savedAt: params.savedAt       };              for (var j = 0; j < headers.length; j++) {         rowData.push(record[headers[j]] || '');       }              if (existRow > 0) {         sheet.getRange(existRow, 1, 1, headers.length).setValues([rowData]);       } else {         sheet.appendRow(rowData);       }     }   }      return ContentService.createTextOutput(JSON.stringify({ status: 'success' }))     .setMimeType(ContentService.MimeType.JSON); } ```
